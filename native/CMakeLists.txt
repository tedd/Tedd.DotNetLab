cmake_minimum_required(VERSION 3.25)
project(DotNetLabProfiler LANGUAGES CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR (NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux"))
  message(FATAL_ERROR "Native profiler requires a supported Windows/Linux x64 build host.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32)
  add_library(Tedd.DotNetLab.Profiler SHARED Tedd.DotNetLab.Profiler/Profiler.cpp Tedd.DotNetLab.Profiler/Profiler.def)
  target_compile_definitions(Tedd.DotNetLab.Profiler PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_compile_options(Tedd.DotNetLab.Profiler PRIVATE /W4 /EHsc)
  target_link_libraries(Tedd.DotNetLab.Profiler PRIVATE ole32 advapi32 bcrypt)
else()
  if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    message(FATAL_ERROR "Linux profiler ABI qualification requires x64.")
  endif()
  find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
  find_package(Threads REQUIRED)
  add_compile_definitions(PAL_STDCPP_COMPAT HOST_64BIT HOST_AMD64 TARGET_64BIT TARGET_AMD64 PLATFORM_UNIX FEATURE_PAL _MIDL_USE_GUIDDEF_)
  include_directories(SYSTEM vendor/coreclr-pal)
  add_library(Tedd.DotNetLab.Profiler SHARED Tedd.DotNetLab.Profiler/LinuxProfiler.cpp vendor/coreclr-pal/corprof_i.cpp)
  set_source_files_properties(vendor/coreclr-pal/corprof_i.cpp PROPERTIES COMPILE_OPTIONS "-Wno-unknown-pragmas")
  set_target_properties(Tedd.DotNetLab.Profiler PROPERTIES PREFIX "" SUFFIX ".so")
  target_compile_options(Tedd.DotNetLab.Profiler PRIVATE -Wall -Wextra -Werror -Wno-pragma-pack)
  target_link_libraries(Tedd.DotNetLab.Profiler PRIVATE OpenSSL::Crypto Threads::Threads)
endif()
target_include_directories(Tedd.DotNetLab.Profiler SYSTEM PRIVATE vendor/coreclr vendor/nlohmann)
enable_testing()
add_executable(RuntimePolicyContract Tedd.DotNetLab.Profiler.Tests/RuntimePolicyContract.cpp)
add_test(NAME native_runtime_policy_contract COMMAND RuntimePolicyContract)
add_executable(HeaderContract Tedd.DotNetLab.Profiler.Tests/HeaderContract.cpp)
target_include_directories(HeaderContract SYSTEM PRIVATE vendor/coreclr)
add_test(NAME native_header_contract COMMAND HeaderContract)
add_executable(CilContract Tedd.DotNetLab.Profiler.Tests/CilContract.cpp)
if(WIN32)
  target_compile_options(CilContract PRIVATE /W4 /EHsc)
else()
  target_compile_options(CilContract PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME native_cil_contract COMMAND CilContract)
add_executable(InliningContract Tedd.DotNetLab.Profiler.Tests/InliningContract.cpp)
if(WIN32)
  target_compile_options(InliningContract PRIVATE /W4 /EHsc)
else()
  target_compile_options(InliningContract PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME native_inlining_contract COMMAND InliningContract)
add_executable(MutationBudgetContract Tedd.DotNetLab.Profiler.Tests/MutationBudgetContract.cpp)
if(WIN32)
  target_compile_options(MutationBudgetContract PRIVATE /W4 /EHsc)
else()
  target_compile_options(MutationBudgetContract PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME native_mutation_budget_contract COMMAND MutationBudgetContract)
